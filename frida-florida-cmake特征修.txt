#!/bin/bash
set -e

echo "===== 1. 同步子模块 ====="
git submodule update --init --recursive

echo "===== 2. 应用 Florida 补丁 ====="
PATCH_BASE="/home/myuser/florida_build_within_GGbonds/Florida-17.15.3/patches"

cd subprojects/frida-core
for patch in "$PATCH_BASE"/frida-core/*.patch; do
    patch -p1 -N < "$patch"
done
cd ../..

cd subprojects/frida-gum
for patch in "$PATCH_BASE"/frida-gum/*.patch; do
    patch -p1 -N < "$patch"
done
cd ../..

echo "===== 3. 使用官方构建脚本编译 ====="
# 注意：这会在 build/ 目录生成产物，并自动处理 Vala
./make          # 或者 python releng/meson.py

echo "===== 编译完成！====="
7.checkout the replace success befor build.
#如果还有 .rej 文件，说明有补丁未完全应用（或手动修复后未删除）。运行以下查找：
find /home/myuser/florida_build_within_GGbonds/frida/subprojects -name "*.rej"

#进入相应子模块，用 --dry-run -N 检查每个补丁是否已经应用（不会修改文件）
cd /home/myuser/florida_build_within_GGbonds/frida/subprojects/frida-core
for patch in /home/myuser/florida_build_within_GGbonds/Florida-17.15.3/patches/frida-core/*.patch; do
    echo "Checking $patch ..."
    patch -p1 --dry-run -N < "$patch"
    if [ $? -eq 0 ]; then
        echo "  ✓ Success (already applied or no changes needed)"
    else
        echo "  ✗ FAILED! This patch is not fully applied."
    fi
done

#对 frida-gum 同样操作

    patch -p1 --dry-run -N < "$patch"

        echo "  ✓ Success (already applied or no changes needed)"
    else
        echo "  ✗ FAILED! This patch is not fully applied."
    fi
done
