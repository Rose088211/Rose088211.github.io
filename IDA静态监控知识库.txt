			用“段内偏移”对抗 ASLR（每次运行时随机基址重置）：
				//TODO ::
				//抢占先机，Hook 初始化函数
				//精确拦截检测线程
				// 1. 获取 so 基址
				var base = Module.findBaseAddress("libtest.so");
				if (base) {
					// 2. 计算实际 Hook 地址
					var targetAddr = base.add(0x1234);
					console.log("[+] Hooking at:", targetAddr);
					
					// 3. 修改内存页属性（仅当目标函数在只读段时需要）
					Memory.protect(targetAddr, 4, 'rwx');

					// 4. 进行 Hook
					Interceptor.attach(targetAddr, {
						onEnter: function(args) { 
							console.log("[+] Function entered. Args:", args[0]); 
						},
						onLeave: function(retval) { 
							console.log("[+] Function left. Ret:", retval);
						}
					});
				} else {
					console.log("[-] libtest.so not found.");
				}
